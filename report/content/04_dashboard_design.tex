\section{Thiết kế bảng điều khiển}
\label{sec:dashboard-design}

\subsection{Kiến trúc tổng quan}

\begin{itemize}
    \item \textit{Giao diện người dùng:} xây dựng bằng React để hiển thị thẻ chỉ số, biểu đồ và quản lý các thao tác tương tác.
    \item \textit{Dịch vụ phía máy chủ:} xây dựng bằng FastAPI để cung cấp dữ liệu và kết quả phân tích cho giao diện.
    \item \textit{Xử lý dữ liệu:} sử dụng các chương trình Python để chuẩn hóa dữ liệu, tính thống kê, thứ hạng và thông tin tóm tắt.
    \item \textit{Lớp dữ liệu:} lưu các bảng PAPI đã chuẩn hóa ở trạng thái chỉ đọc và dùng chung cho các trang.
    \item \textit{Trợ lý AI:} hoạt động độc lập với luồng hiển thị và chỉ thực thi phân tích sau khi được phê duyệt.
\end{itemize}

\subsection{Bố cục và luồng khám phá dữ liệu}

\begin{itemize}
    \item \textit{Điều hướng:} năm trang được sắp xếp từ Tổng quan đến bốn hướng phân tích chuyên sâu.
    \item \textit{Bố cục chung:} tiêu đề, bộ lọc, bốn thẻ chỉ số và bốn biểu đồ trên lưới 2 $\times$ 2.
    \item \textit{Nhận xét:} đặt ngay dưới biểu đồ tương ứng để liên kết kết quả với hình vẽ.
    \item \textit{Thích ứng màn hình:} lưới hai cột trên màn hình lớn và một cột khi chiều rộng giảm.
\end{itemize}

\subsection{Màu sắc và nguyên tắc mã hóa trực quan}

\begin{itemize}
    \item \textit{Giao diện:} nền xanh xám nhạt, thẻ trắng, chữ xanh than và màu xanh ngọc làm điểm nhấn.
    \item \textit{Tính nhất quán:} mỗi vùng và mỗi lĩnh vực giữ một màu cố định trên toàn bộ bảng điều khiển.
    \item \textit{Thang màu:} tuần tự cho mức điểm; phân kỳ có tâm tại 0 cho mức thay đổi và tương quan.
    \item \textit{Khả năng phân biệt:} kết hợp màu với dấu, hướng, độ đậm nét, ký hiệu và nhãn trực tiếp.
    \item \textit{Dữ liệu thiếu:} dùng màu xám nhạt và không xếp vào nhóm điểm thấp.
\end{itemize}

\subsection{Bộ lọc, tương tác và chế độ phóng to biểu đồ}

\begin{itemize}
    \item \textit{Bộ lọc:} phạm vi sáu hoặc tám lĩnh vực, năm, vùng, tỉnh và cặp lĩnh vực tùy theo từng trang.
    \item \textit{Trạng thái:} lựa chọn hiện tại được hiển thị rõ và giữ lại khi làm mới hoặc chia sẻ đường dẫn.
    \item \textit{Di chuột:} làm nổi đối tượng và hiển thị chú giải chi tiết.
    \item \textit{Lựa chọn:} giữ đối tượng đang quan tâm và cập nhật các biểu đồ liên quan.
    \item \textit{Hỗ trợ đọc dữ liệu:} cung cấp bảng tóm tắt cho giá trị chính xác và thao tác bằng bàn phím.
    \item \textit{Phóng to:} mở cùng biểu đồ, nhận xét, đơn vị và số quan sát trong chế độ xem riêng ở Hình~\ref{fig:dashboard-chart-focus}.
\end{itemize}

\subsection{Trợ lý AI nổi và cơ chế phê duyệt của người dùng}

\begin{itemize}
    \item \textit{Vị trí:} nút mở cố định ở góc dưới bên phải của cả năm trang, như Hình~\ref{fig:dashboard-floating-assistant}.
    \item \textit{Ngữ cảnh:} ghi nhận trang, bộ lọc và đối tượng đang chọn tại thời điểm gửi yêu cầu.
    \item \textit{Câu hỏi kiến thức:} trả lời ngắn gọn và kèm nguồn.
    \item \textit{Yêu cầu phân tích:} hiển thị giải thích và toàn bộ mã lệnh ở trạng thái chờ duyệt.
    \item \textit{Phê duyệt:} người dùng yêu cầu chỉnh lại hoặc đồng ý chạy phiên bản mới nhất tại máy.
    \item \textit{Kết quả:} trả về bảng, biểu đồ hoặc thông báo lỗi trong cùng phiên hội thoại.
\end{itemize}
